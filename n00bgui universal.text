    local function initializeSystem()
        local char = player.Character or player.CharacterAdded:Wait()
        isCrouchingR6, isPunchingR6, isCrouchingR15, isPunchingR15 = false, false, false, false
        origRootC0_R6, origLeftHipC0_R6, origRightHipC0_R6, origNeckC0_R6, origRightShoulderC0_R6 = nil, nil, nil, nil, nil
        origRootC0_R15, origLeftHipC0_R15, origRightHipC0_R15, origNeckC0_R15, origRightShoulderC0_R15 = nil, nil, nil, nil, nil
        origRightElbowC0_R15, origRightWristC0_R15 = nil, nil
        if freezeConnectionR15 then freezeConnectionR15:Disconnect() end
        
        currentRig = getRigType(char)
        print("[Minecraft Unificado]: Detectado avatar de tipo " .. currentRig)
        
        local animateScript = char:WaitForChild("Animate", 5)
        if animateScript then 
            animateBackup = animateScript:Clone()
            animateScript.Disabled = true 
            animateScript:Destroy() 
        end
        
        local humanoid = char:WaitForChild("Humanoid")
        local animator = humanoid:FindFirstChildOfClass("Animator") or humanoid
        if animator then
            for _, track in pairs(animator:GetPlayingAnimationTracks()) do track:Stop() end
        end
        
        setupJoints(char, currentRig)
        handleMinecraftDeath(char, humanoid, currentRig)
        
        if currentRig == "R15" then
            jointsToFreezeR15 = {
                char:WaitForChild("RightLowerArm", 5) and char.RightLowerArm:WaitForChild("RightElbow", 5),
                char:WaitForChild("RightHand", 5) and char.RightHand:WaitForChild("RightWrist", 5),
                char:WaitForChild("LeftLowerArm", 5) and char.LeftLowerArm:WaitForChild("LeftElbow", 5),
                char:WaitForChild("LeftHand", 5) and char.LeftHand:WaitForChild("LeftWrist", 5),
                char:WaitForChild("RightLowerLeg", 5) and char.RightLowerLeg:WaitForChild("RightKnee", 5),
                char:WaitForChild("RightFoot", 5) and char.RightFoot:WaitForChild("RightAnkle", 5),
                char:WaitForChild("LeftLowerLeg", 5) and char.LeftLowerLeg:WaitForChild("LeftKnee", 5),
                char:WaitForChild("LeftFoot", 5) and char.LeftFoot:WaitForChild("LeftAnkle", 5)
            }
            freezeConnectionR15 = RunService.Stepped:Connect(function()
                if not char or not char.Parent or humanoid.Health <= 0 then 
                    freezeConnectionR15:Disconnect() 
                    return 
                end
                for _, joint in pairs(jointsToFreezeR15) do
                    if isPunchingR15 and (joint.Name == "RightElbow" or joint.Name == "RightWrist") then continue end
                    if joint and joint:IsA("Motor6D") then
                        joint.C0 = CFrame.new(joint.C0.Position) * CFrame.Angles(0, 0, 0)
                    end
                end
            end)
        end
        updateButtonColor()
    end
    

    local function restoreAnimations()
        local char = player.Character
        if not char then return end
        
        if freezeConnectionR15 then 
            freezeConnectionR15:Disconnect() 
            freezeConnectionR15 = nil
        end
        
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if humanoid then
            if currentRig == "R6" and isCrouchingR6 then
                humanoid.WalkSpeed = speedSavedR6
            elseif currentRig == "R15" and isCrouchingR15 then
                humanoid.WalkSpeed = speedSavedR15
            end
        end
        
        if currentRig == "R6" then
            local torso = char:FindFirstChild("Torso")
            local rootPart = char:FindFirstChild("HumanoidRootPart")
            if torso and rootPart then
                local rootJoint = rootPart:FindFirstChild("RootJoint")
                local leftHip = torso:FindFirstChild("Left Hip")
                local rightHip = torso:FindFirstChild("Right Hip")
                local neck = torso:FindFirstChild("Neck")
                local rightShoulder = torso:FindFirstChild("Right Shoulder")
                
                if rootJoint and origRootC0_R6 then rootJoint.C0 = origRootC0_R6 end
                if leftHip and origLeftHipC0_R6 then leftHip.C0 = origLeftHipC0_R6 end
                if rightHip and origRightHipC0_R6 then rightHip.C0 = origRightHipC0_R6 end
                if neck and origNeckC0_R6 then neck.C0 = origNeckC0_R6 end
                if rightShoulder and origRightShoulderC0_R6 then rightShoulder.C0 = origRightShoulderC0_R6 end
            end
            isCrouchingR6 = false
            isPunchingR6 = false
        elseif currentRig == "R15" then
            local lowerTorso = char:FindFirstChild("LowerTorso")
            local upperTorso = char:FindFirstChild("UpperTorso")
            local rightUpperArm = char:FindFirstChild("RightUpperArm")
            if lowerTorso and upperTorso and rightUpperArm then
                local rootJoint = lowerTorso:FindFirstChild("Root")
                local leftHip = char:FindFirstChild("LeftUpperLeg") and char.LeftUpperLeg:FindFirstChild("LeftHip")
                local rightHip = char:FindFirstChild("RightUpperLeg") and char.RightUpperLeg:FindFirstChild("RightHip")
                local neck = upperTorso:FindFirstChild("Head")
                local rightShoulder = rightUpperArm:FindFirstChild("RightShoulder")
                local rightElbow = char:FindFirstChild("RightLowerArm") and char.RightLowerArm:FindFirstChild("RightElbow")
                local rightWrist = char:FindFirstChild("RightHand") and char.RightHand:FindFirstChild("RightWrist")
                
                if rootJoint and origRootC0_R15 then rootJoint.C0 = origRootC0_R15 end
                if leftHip and origLeftHipC0_R15 then leftHip.C0 = origLeftHipC0_R15 end
                if rightHip and origRightHipC0_R15 then rightHip.C0 = origRightHipC0_R15 end
                if neck and origNeckC0_R15 then neck.C0 = origNeckC0_R15 end
                if rightShoulder and origRightShoulderC0_R15 then rightShoulder.C0 = origRightShoulderC0_R15 end
                if rightElbow and origRightElbowC0_R15 then rightElbow.C0 = origRightElbowC0_R15 end
                if rightWrist and origRightWristC0_R15 then rightWrist.C0 = origRightWristC0_R15 end
            end
            isCrouchingR15 = false
            isPunchingR15 = false
        end
        
        
        if animateBackup then
            local newAnimate = animateBackup:Clone()
            newAnimate.Parent = char
            newAnimate.Disabled = false
        end
    end
    
    ButtonCrouch.MouseButton1Click:Connect(function()

        end
    end)

    end)
